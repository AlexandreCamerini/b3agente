  {"ticker", "symbol", "source", "providerStatus": "ok"|"degraded",
   "underlyingPrice": <float|None>, "expirations": [...], "expiration": <str>,

(`server/app/opcoes_lastreadas.py:20,42-56`):

  _LIQUIDEZ_MINIMA = 40
  _candidato_valido(c): lastPrice numérico e > 0 E liquidity_score(...)["score"] >= 40
  _escolher_contrato(candidatos, melhor): melhor(validos, key=strike or 0),
    onde `melhor` é `min` (menor strike = call OTM mais próxima) ou
    `max` (maior strike = put mais próxima do spot)
